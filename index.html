from playwright.sync_api import sync_playwright, TimeoutError

def execute_qa_workflow(target_url: str, max_retries: int = 3) -> dict:
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        page = browser.new_page()
        for attempt in range(1, max_retries + 1):
            try:
                resp = page.goto(target_url, wait_until="networkidle", timeout=10000)
                assert resp.status == 200, f"HTTP_ERR_{resp.status}"
                return {"status": "PASS", "attempt": attempt}
            except (TimeoutError, AssertionError) as err:
                if attempt == max_retries:
                    page.screenshot(path="fault_capture.png")
                    return {"status": "FAIL", "error": str(err)}
